## Single sigmoid neuron

```python
# neuron_sketch.py — original structure
import math

def sigmoid(z):
    return 1.0 / (1.0 + math.exp(-z))

class Neuron:
    def __init__(self, weights, bias=0.0):
        self.weights = list(weights)
        self.bias = bias

    def feedforward(self, inputs):
        s = self.bias + sum(w * x for w, x in zip(self.weights, inputs))
        return sigmoid(s)

# n = Neuron([-1.0, 1.0], 0.0); print(n.feedforward([12.0, 8.0]))  # ≈ 0.017986
```

## Shallow 2–2–1 net

```python
# shallow_net_sketch.py — original structure
class NeuralNetwork:
    def __init__(self):
        w = [0.0, 1.0]
        self.h1 = Neuron(w)
        self.h2 = Neuron(w)
        self.out = Neuron(w)

    def feedforward(self, x):
        a = self.h1.feedforward(x)
        b = self.h2.feedforward(x)
        return self.out.feedforward([a, b])

# network.feedforward([2.0, 3.0]) ≈ 0.7216
```
