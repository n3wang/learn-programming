  <Problem
    type="concept"
    title="12.5b.W2"

    keywords={['measure', 'probability', 'unitary']}
    prompt={`Why is the M (measure) operation not a quantum gate in the same sense as H or X?`}
    answer={`Gates are unitary and reversible; measurement collapses the state and yields classical bits, so it is not unitary.`}
  />
  <Problem
    type="concept"
    title="12.5b.W3"

    keywords={['toffoli', 'cnot', 'carry', 'sum']}
    prompt={`In the three-qubit half-adder, which gate writes the carry and which writes the sum?`}
    answer={`Toffoli writes the AND carry into the third qubit; CNOT writes the XOR sum into the second qubit.`}
