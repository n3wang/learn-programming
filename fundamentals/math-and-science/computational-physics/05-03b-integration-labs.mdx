  hint="h=1/N; f_i=exp(-i*h). Trap: h*(0.5*f0+sum(f1..f_{N-1})+0.5*fN). Simpson: h/3*(f0+4*odds+2*evens+fN)."
  solution={`import math

def trap_simp(N):
    h = 1.0 / N
    fs = [math.exp(-i * h) for i in range(N + 1)]
    trap = h * (0.5 * fs[0] + sum(fs[1:N]) + 0.5 * fs[N])

```python
# quad_ref.py
import math

def left_sum(f, a, b, N):
    h = (b - a) / N
    return h * sum(f(a + i * h) for i in range(N))

def mid_sum(f, a, b, N):
    h = (b - a) / N
    return h * sum(f(a + (i + 0.5) * h) for i in range(N))

def trap(f, a, b, N):
    h = (b - a) / N
    s = 0.5 * (f(a) + f(b))
    for i in range(1, N):
        s += f(a + i * h)
    return s * h

def simpson(f, a, b, N):
    h = (b - a) / N
    s = f(a) + f(b)
    for i in range(1, N, 2):
        s += 4 * f(a + i * h)
    for i in range(2, N, 2):
        s += 2 * f(a + i * h)
    return s * h / 3

def romberg(f, a, b, N):
    return (4 * trap(f, a, b, 2 * N) - trap(f, a, b, N)) / 3
```
