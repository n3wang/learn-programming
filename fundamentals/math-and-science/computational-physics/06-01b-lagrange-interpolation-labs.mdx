  sampleLog={`(input) 4
0 -12
1 -12

4 -60
0.5
-10.125000`}
  starter={`def lagrange(nodes, x):
    # nodes: list of (xi, gi)
    return 0.0
`}
  wrapSuffix={`
n = int(input())
nodes = [tuple(map(float, input().split())) for _ in range(n)]
x = float(input())
print(f"{lagrange(nodes, x):.6f}")
`}
  hint="For each i, multiply (x-xj)/(xi-xj) over all j != i, then sum gi * that product."
  solution={`def lagrange(nodes, x):
    n = len(nodes)
    total = 0.0
    for i in range(n):
        xi, gi = nodes[i]
        term = gi
        for j in range(n):
            if j == i:
                continue
            xj, _ = nodes[j]
            term *= (x - xj) / (xi - xj)
        total += term
    return total
`}
  sourceChecks={[
    {name: 'nested loop over nodes', pattern: 'for .* in .*:\\s*\\n.*for', must: false, hint: 'product over j != i inside sum over i'},
  ]}
  tests={[
    {
      name: 'cubic check (6.24)-(6.26)',
      stdin: '4\n0 -12\n1 -12\n2 -24\n4 -60\n0.5',

  hint="Find the interval E falls in (index k with TABLE[k][0] <= E < TABLE[k+1][0]), then use TABLE[k-1], TABLE[k], TABLE[k+1] as the 3 nearest nodes."
  solution={`TABLE = [
    (0, 10.6), (25, 16.0), (50, 45.0), (75, 83.5), (100, 52.8),
    (125, 19.9), (150, 10.8), (175, 8.25), (200, 4.7),

def lagrange(nodes, x):
    n = len(nodes)
    total = 0.0
    for i in range(n):
        xi, gi = nodes[i]
        term = gi
        for j in range(n):
            if j == i:
                continue
            xj, _ = nodes[j]
            term *= (x - xj) / (xi - xj)

def local_interp(E):
    k = 0
    for i in range(len(TABLE) - 1):
        if TABLE[i][0] <= E < TABLE[i + 1][0]:
            k = i
            break
    lo = max(0, k - 1)
    hi = min(len(TABLE) - 1, lo + 2)
    lo = hi - 2
    nodes = TABLE[lo:hi + 1]
    return lagrange(nodes, E)
