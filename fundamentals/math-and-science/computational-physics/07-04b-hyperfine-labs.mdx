def build_V(W=1.0):
    return [
        [W, 0, 0, 0],
        [0, -W, 2 * W, 0],
        [0, 2 * W, -W, 0],
        [0, 0, 0, W],
    ]

def spectrum(W=1.0):
    return sorted([W, W, W, -3 * W])  # ΔE/W = 4
