<CodeExercise
  title="MAD frames"
  heading="exercise-cp-119-diff"
  lang="python"
  filename="main.py"
  prompt="mean abs diff; 10 decimals."
  sampleLog={`6.6666666667`}

`}
  wrapSuffix={`

`}
  hint="Average of abs differences."
  solution={`f1 = [100.0, 100.0, 100.0]
f2 = [100.0, 110.0, 90.0]

`}
  sourceChecks={[
    {name: 'abs', pattern: 'abs', must: true, hint: 'absolute differences'},
  ]}
  tests={[
    {name: 'mad', equals: '6.6666666667'},
  ]}
/>

</Exercise>
