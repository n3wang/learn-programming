  <Problem
    type="concept"
    title="12.9b.W1"
    minWords={3}
    keywords={['oracle', 'phase', 'mark']}
    prompt={`What does Grover’s oracle do to the marked basis state?`}
    answer={`It multiplies that state’s amplitude by −1 (a phase flip), leaving unmarked states unchanged.`}
  />
  <Problem
    type="concept"
    title="12.9b.W2"
    minWords={3}
    keywords={['diffuser', 'mean', 'reflect']}
    prompt={`In one sentence, what does the diffuser Uψ do to amplitudes?`}
    answer={`It reflects each amplitude about the mean amplitude (inversion about the average).`}
  />
  <Problem
    type="concept"
    title="12.9b.W3"
    minWords={3}
    keywords={['noise', 'hardware', 'simulator']}
    prompt={`Why might IBM hardware fail to show a clean |1001⟩ peak that the simulator shows?`}
    answer={`Gate and readout noise smear probability into other bitstrings; the ideal simulator has no such noise.`}
  />
